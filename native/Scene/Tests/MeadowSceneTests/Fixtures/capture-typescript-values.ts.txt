// Captures values from the web version's TypeScript, for the native tests to check against.
const THREE = await import(`${process.argv[2]}/node_modules/three/build/three.module.js`);
// No browser here: props.ts draws its textures on canvases when loaded, which these values never use.
const ctx = new Proxy({}, { get: (_t, k) => (k === 'measureText' ? () => ({ width: 10 }) : k === 'createRadialGradient' || k === 'createLinearGradient' ? () => ({ addColorStop() {} }) : () => {}), set: () => true });
(globalThis as any).document = { createElement: () => ({ width: 0, height: 0, getContext: () => ctx }) };
const root = process.argv[2];
const kit = await import(`${root}/src/meadow/kit.ts`);
const { sculpt, PIVOT, ARM_BASE, ARM_MIN, ARM_MAX } = await import(`${root}/src/cat/sculpt.ts`);
const { chance, settle } = await import(`${root}/src/village/scatter.ts`);
const { DAISY, LUPIN, MUSHROOM, STONE } = await import(`${root}/src/meadow/flora.ts`);
const { STATES, relaxed } = await import(`${root}/src/cat/states.ts`);

const r = kit.seeded(20261112);
const seeded = Array.from({ length: 5 }, () => r());
const c = chance('sim-shop-mwditcdj');
const chanced = Array.from({ length: 3 }, () => c());
const s = chance('sim-shop-mwditcdj/main');
const placed = [{ x: 0, z: 0, reach: 1.35 }, { x: 3, z: 1, reach: 1 }];
const settled = settle(s, placed, 1, 0.65, 0, 0, 2, 4);

const form = sculpt();
const pick = [0, 100, 1234, 3000, 5000, form.count - 1];
const rest = pick.map((i) => [form.rest[i * 3], form.rest[i * 3 + 1], form.rest[i * 3 + 2]]);
const weights = pick.map((i) => [form.earPull[0][i], form.earPull[1][i], form.armPull[0][i], form.armPull[1][i], form.turn[i]]);
const eyes = form.eyes.map((e) => [e.position.x, e.position.y, e.position.z, e.facing.x, e.facing.y, e.facing.z, e.facing.w]);

// Cat.ts pose(), for one set of inputs.
const P = { yaw: 0.31, pitch: -0.2, roll: 0.12, earFlat: 0.5, earPerk: 0.3, flickL: 0.25, flickR: -0.1, pushL: [-0.1, 0.2, 0.4], pushR: [0.2, -0.3, 0.35] };
function posed(i: number) {
  let x = form.rest[i * 3], y = form.rest[i * 3 + 1] - PIVOT, z = form.rest[i * 3 + 2];
  const eL = form.earPull[0][i], eR = form.earPull[1][i];
  x += (eR - eL) * P.earFlat * 0.22 + (eR * P.flickR - eL * P.flickL) * 0.16;
  y -= (eL + eR) * P.earFlat * 0.26;
  z += (eL * P.flickL + eR * P.flickR) * 0.1 + (eL + eR) * P.earPerk * 0.1;
  const aL = form.armPull[0][i], aR = form.armPull[1][i];
  x += aL * P.pushL[0] + aR * P.pushR[0]; y += aL * P.pushL[1] + aR * P.pushR[1]; z += aL * P.pushL[2] + aR * P.pushR[2];
  const t = form.turn[i];
  const rr = P.roll * t, a = P.pitch * t, b = P.yaw * t;
  const x0 = x * Math.cos(rr) - y * Math.sin(rr), y0 = x * Math.sin(rr) + y * Math.cos(rr);
  const y1 = y0 * Math.cos(a) - z * Math.sin(a), z1 = y0 * Math.sin(a) + z * Math.cos(a);
  return [x0 * Math.cos(b) + z1 * Math.sin(b), y1 + PIVOT, -x0 * Math.sin(b) + z1 * Math.cos(b)];
}
const pose = pick.map(posed);

// What each state wants at a few moments.
const fake = { hands: [new THREE.Vector3(), new THREE.Vector3()], wards: [], presses: 0, pressButton() {}, thump() {}, jump() {} };
const states: Record<string, unknown> = {};
for (const name of Object.keys(STATES)) {
  for (const t of [0.3, 2.7]) {
    const w = relaxed();
    fake.presses = 0;
    STATES[name].act(w, t, fake);
    states[`${name}@${t}`] = {
      aim: Array.isArray(w.aim) ? w.aim : typeof w.aim === 'string' ? w.aim : 'spot', look: w.look, lid: w.lid, eye: w.eye, pupil: w.pupil,
      earFlat: w.earFlat, earPerk: w.earPerk, tail: w.tail, tailSpeed: w.tailSpeed, tailAmp: w.tailAmp, tailPuff: w.tailPuff,
      tall: w.tall, lie: w.lie, prone: w.prone, breath: w.breath, shiver: w.shiver, sway: w.sway, prop: w.prop, bubble: w.bubble,
      hands: w.hands.map((h: any) => (h ? [h.x, h.y, h.z] : null)),
    };
  }
}

const geometry = (g: any) => { g.computeBoundingBox(); const b = g.boundingBox!; return { vertices: g.attributes.position.count, triangles: (g.index ? g.index.count : g.attributes.position.count) / 3, min: b.min.toArray(), max: b.max.toArray() }; };
const out = {
  seeded, chanced, settled, heightAt: [[100, 50, null], [0, 0, null], [-80, 30, null]].map(([x, z]) => [x, z, kit.fbm(x / 90, z / 90, 3)]),
  fbm: [kit.fbm(1.3, 2.7, 21), kit.fbm(-5.1, 0.25, 3)], noise: [kit.noise(3.3, -1.7, 0), kit.noise(12.9, 4.4, 13)],
  sculpt: { count: form.count, triangles: form.index.count / 3, pick, rest, weights, eyes, armBase: ARM_BASE.map((v: any) => v.toArray()), armMin: ARM_MIN, armMax: ARM_MAX },
  pose: { inputs: P, pick, posed: pose },
  states,
  flora: { daisy0: geometry(DAISY[0].shape), lupin1: geometry(LUPIN[1].shape), mushroom0: geometry(MUSHROOM[0].shape), stone1: geometry(STONE[1].shape), daisyTop: DAISY[0].top.toArray() },
};
console.log(JSON.stringify(out));
